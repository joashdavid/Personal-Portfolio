A glimpse of my AI-Powered Automation Framework. Using Playwright and Python, tests are dynamically generated and locators heal themselves when DOM elements shift.

  • → AI-assisted locator healing

test_snooker_platform.py 
import pytest
from framework.pages.login_page import LoginPage
from framework.ai.locator_healer import Healer

# AI-assisted self-healing test case
@pytest.mark.regression
def test_tournament_bracket_generation(page, db):
    login = LoginPage(page)
    login.navigate()
    login.authenticate("admin@snooker.dev", "secure_password")

    # If selector fails, AI Healer inspects DOM and updates the locator
    with Healer(page, "#generate-bracket-btn"):
        page.click("#generate-bracket-btn")

    # Assert bracket is accurately generated in PostgreSQL
    bracket_count = db.query("SELECT COUNT(*) FROM tournament_brackets")
    assert bracket_count == 16
